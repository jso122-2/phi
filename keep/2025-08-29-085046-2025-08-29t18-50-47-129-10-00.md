# dawn_core/snapshot.py
import json, time, pathlib
from dawn_core.state import get_state
SNAP_DIR = pathlib.Path("runtime/snapshots"); SNAP_DIR.mkdir(parents=True, exist_ok=True)

def snapshot(tag="auto"):
    s = get_state()
    path = SNAP_DIR / f"{int(time.time())}_{tag}.json"
    path.write_text(json.dumps(s.__dict__, indent=2))
    return str(path)

def restore(path: str):
    from dawn_core.state import set_state
    data = json.loads(pathlib.Path(path).read_text())
    set_state(**data)
