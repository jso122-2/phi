0) mission recap (one paragraph)

Thinkerbell takes a short brief (or semi‑structured text), picks the most likely legal template (from your training set), extracts key fields (client, fee, deliverables, terms, etc.), and renders a finished contract in HTML (and optional .docx) that looks like the brand’s template. Retrieval helps with template selection; rendering fills a template skeleton with extracted fields. For the demo, we prioritize deterministic, CPU‑only operation with graceful fallbacks (no FAISS/GPU required).

api/routers/search.py

from fastapi import APIRouter
from pydantic import BaseModel
from typing import Optional, Dict

router = APIRouter()

class SearchIn(BaseModel):
    text: str
    k: int = 5
    filters: Optional[Dict[str,str]] = None

@router.post("/search")
def search(body: SearchIn):
    from api.services.embedding_service import EMBEDDER
    from api.services.retrieval_service import RETRIEVER
    vec = EMBEDDER.embed(body.text)
    hits, timing = RETRIEVER.search(vec, k=body.k, filters=body.filters or {})
    return {"hits": hits, "timing": timing, "used": {"main": RETRIEVER.index_type!="sklearn", "efSearch": None}}

api/routers/generate.py

from fastapi import APIRouter
from pydantic import BaseModel
from typing import Optional

router = APIRouter()

class GenIn(BaseModel):
    text: str
    preferred_template: Optional[str] = None

@router.post("/generate-document")
def generate_document(body: GenIn):
    from api.services.embedding_service import EMBEDDER
    from api.services.retrieval_service import RETRIEVER
    from api.services.generation_service import GENERATOR

    vec = EMBEDDER.embed(body.text)
    template = body.preferred_template or RETRIEVER.pick_template(vec)
    fields = GENERATOR.extract_fields(body.text, template_hint=template)
    html = GENERATOR.render(template, fields)
    return {
        "template": template,
        "fields": fields,
        "html": html,
        "plain_text": GENERATOR.to_text(html)
    }

api/services/embedding_service.py

import os, numpy as np
from sentence_transformers import SentenceTransformer

class _Embedder:
    def __init__(self):
        model_dir = os.getenv("THINKERBELL_MODEL_DIR", "models/thinkerbell-encoder-best")
        self.model = SentenceTransformer(model_dir)
        self.device = str(self.model._target_device)
    def embed(self, text: str):
        v = self.model.encode([text], show_progress_bar=False, normalize_embeddings=True)
        return np.asarray(v, dtype="float32")[0]

EMBEDDER = _Embedder()

api/services/retrieval_service.py

import os, json, time, numpy as np
from sklearn.neighbors import NearestNeighbors

class _Retriever:
    def __init__(self):
        data_dir = os.getenv("THINKERBELL_DATA_DIR", ".")
        meta_path = os.path.join(data_dir, "artifacts", "embeddings", "metadata.jsonl")
        vec_path = os.path.join(data_dir, "artifacts", "embeddings", "encoder_vectors.npy")

        # fallback if embeddings not built yet – use dataset sample texts as corpus
        self.meta = []
        self.vecs = None
        if os.path.exists(vec_path) and os.path.exists(meta_path):
            self.vecs = np.load(vec_path).astype("float32")
            with open(meta_path, "r", encoding="utf-8") as f:
                self.meta = [json.loads(l) for l in f]
        else:
            # lightweight corpus from dataset (first 1k)
            from core.data_loader import load_all_samples
            samples = load_all_samples(limit=1000)
            from api.services.embedding_service import EMBEDDER
            self.meta = [{"id": i, "sample_id": s["sample_id"],
                          "template": s["template_mapping"]["best_template_match"],
                          "path": s.get("_path", "")} for i,s in enumerate(samples)]
            self.vecs = np.stack([EMBEDDER.embed(s["raw_input"]["text"]) for s in samples]).astype("float32")

        self.nn = NearestNeighbors(metric="cosine", n_neighbors=50).fit(self.vecs)
        self.index_type = "sklearn"

    def size(self): return len(self.meta)
    def templates(self): return sorted(list({m["template"] for m in self.meta}))

    def search(self, q, k=5, filters=None):
        t0 = time.time()
        dists, idxs = self.nn.kneighbors([q], n_neighbors=min(50, len(self.meta)))
        hits = []
        for i, dist in zip(idxs[0], dists[0]):
            m = self.meta[i]
            if filters:
                ok = True
                for fk, fv in filters.items():
                    if fk == "template" and fv not in m["template"]:
                        ok = False; break
                if not ok: continue
            hits.append({
                "id": int(m["id"]), "score": float(1.0 - dist),
                "template": m["template"], "path": m.get("path",""),
                "sample_id": m.get("sample_id","")
            })
            if len(hits) >= k: break
        return hits, {"embed_ms": 0, "search_ms": int((time.time()-t0)*1000)}

    def pick_template(self, q):
        hits, _ = self.search(q, k=10, filters=None)
        return hits[0]["template"] if hits else "standard_agreement_v1"

RETRIEVER = _Retriever()

api/services/extraction_rules.py

import re
def extract(text: str):
    # ultra-simple patterns – deterministic demo-safe
    money = re.search(r"(\$\s?\d[\d,\.]*\s?(?:AUD|USD)?)", text, re.I)
    months = re.search(r"(\b\d+\s*months?\b)", text, re.I)
    weeks = re.search(r"(\b\d+\s*weeks?\b)", text, re.I)
    territory = re.search(r"\b(Australia|New Zealand|United States|UK)\b", text, re.I)
    client = re.search(r"(?:between|with|on behalf of)\s+([A-Z][A-Za-z0-9\-\s&\.]+)", text)
    return {
        "client": (client.group(1).strip() if client else None),
        "fee": (money.group(1) if money else None),
        "engagement_term": (months.group(1) if months else None),
        "exclusivity_period": (weeks.group(1) if weeks else None),
        "territory": (territory.group(1) if territory else None)
    }

api/services/generation_service.py

import os, re
from bs4 import BeautifulSoup
from jinja2 import Template
from api.services.extraction_rules import extract as rule_extract

class _Generator:
    def __init__(self):
        self.tpl_dir = os.getenv("THINKERBELL_TEMPLATES_DIR", "artifacts/template_skeletons")
        self.tpl_cache = {}
    def _load_template_html(self, template_id: str):
        # map template_id to a safe filename (very light mapping)
        name = re.sub(r"[^A-Za-z0-9_\-]+", "_", template_id).strip("_").lower()
        path = os.path.join(self.tpl_dir, f"{name}.html")
        if not os.path.exists(path):
            path = os.path.join(self.tpl_dir, "standard_agreement_v1.html")
        if path not in self.tpl_cache:
            with open(path, "r", encoding="utf-8") as f:
                self.tpl_cache[path] = Template(f.read())
        return self.tpl_cache[path]
    def extract_fields(self, text: str, template_hint: str):
        base = rule_extract(text)
        # extend with obvious list-like extractions
        deliverables = []
        for line in text.splitlines():
            if re.search(r"\b(instagram|post|story|reel|deliverable)\b", line, re.I):
                deliverables.append(line.strip("-• ").strip())
        if deliverables:
            base["deliverables"] = deliverables[:6]
        return base
    def render(self, template_id: str, fields: dict):
        tpl = self._load_template_html(template_id)
        return tpl.render(**(fields or {}))
    def to_text(self, html: str):
        return BeautifulSoup(html, "html.parser").get_text("\n", strip=True)

GENERATOR = _Generator()

core/data_loader.py

import os, json, glob

def load_all_samples(root="synthetic_dataset/samples", limit=None):
    files = sorted(glob.glob(os.path.join(root, "**", "tb_*.json"), recursive=True))
    out = []
    for i, fp in enumerate(files):
        try:
            with open(fp, "r", encoding="utf-8") as f:
                j = json.load(f)
                j["_path"] = fp
                out.append(j)
        except Exception:
            continue
        if limit and len(out) >= limit: break
    return out
