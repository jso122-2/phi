3) Patch builder — write the change into a sandbox copy

import re, shutil, pathlib, uuid
from dataclasses import dataclass

@dataclass
class PatchResult:
    run_id: str
    sandbox_dir: str
    target_rel: str
    applied: bool
    reason: str = ""

def make_sandbox(proposal) -> PatchResult:
    run_id = str(uuid.uuid4())[:8]
    root = pathlib.Path("sandbox_mods") / run_id
    root.mkdir(parents=True, exist_ok=True)
    src = pathlib.Path(proposal.target)
    dst = root / src.relative_to("dawn_core")
    dst.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(src, dst)
    text = dst.read_text()
    new = re.sub(r"(DU_STEP\s*=\s*)([0-9.]+)", rf"\\g<1>{proposal.proposed_value}", text, count=1)
    if new == text:
        return PatchResult(run_id, str(root), str(dst), False, "DU_STEP constant not found")
    dst.write_text(new)
    return PatchResult(run_id, str(root), str(dst), True, "")

import json, os, subprocess, sys, pathlib

def run_sandbox(run_id: str, sandbox_dir: str, ticks: int = 30) -> dict:
    env = os.environ.copy()
    project = str(pathlib.Path(".").resolve())
    env["PYTHONPATH"] = f"{sandbox_dir}:{project}:{env.get('PYTHONPATH','')}"
    code = f'''
import json
from dawn_core.state import set_state, get_state, label_for
from dawn_core.tick_orchestrator import demo_step
set_state(unity=0.60, awareness=0.60, momentum=0.01, level='coherent', ticks=0)
h=[]
for i in range({ticks}):
    s=get_state()
    u=min(1.0, s.unity + demo_step())
    a=min(1.0, s.awareness + demo_step())
    set_state(unity=u, awareness=a, level=label_for(u,a), ticks=s.ticks+1)
    h.append((s.ticks, u, a))
res={{"ticks":len(h),"start_unity":h[0][1],"end_unity":h[-1][1],"delta_unity":h[-1][1]-h[0][1],"end_level":get_state().level}}
print(json.dumps(res))
'''
    p = subprocess.run([sys.executable, "-c", code], env=env, capture_output=True, text=True)
    if p.returncode != 0:
        return {"ok": False, "error": p.stderr}
    return {"ok": True, "result": json.loads(p.stdout)}

5) Policy gate — accept only if better + safe

Prompt to Cursor:

Create dawn_core/self_mod/policy_gate.py:

from dataclasses import dataclass

@dataclass
class GateDecision:
    accept: bool
    reason: str

def decide(baseline: dict, sandbox: dict) -> GateDecision:
    if not sandbox.get("ok"):
        return GateDecision(False, f"sandbox error: {sandbox.get('error')}")
    sb = sandbox["result"]
    # safety floors
    if sb["end_unity"] < 0.85:
        return GateDecision(False, "unity below safety floor")
    # improvement vs baseline
    if sb["delta_unity"] <= baseline.get("delta_unity", 0.0) + 0.01:
        return GateDecision(False, "no meaningful improvement")
    if sb["end_level"] not in ("meta_aware","transcendent"):
        return GateDecision(False, f"end level inadequate: {sb['end_level']}")
    return GateDecision(True, "passes thresholds")
