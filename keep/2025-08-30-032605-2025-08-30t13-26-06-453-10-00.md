# dawn_core/self_mod/patch_builder.py
import pathlib, uuid, shutil
from dataclasses import dataclass

@dataclass
class PatchResult:
    run_id: str
    sandbox_dir: str
    applied: bool
    reason: str = ""

TEMPLATE = """from dataclasses import dataclass

@dataclass
class Params:
    DU_STEP: float = {du_step}

params = Params()
"""

def make_sandbox_du_step(new_value: float) -> PatchResult:
    run_id = str(uuid.uuid4())[:8]
    root = pathlib.Path("sandbox_mods") / run_id / "dawn_core"
    root.mkdir(parents=True, exist_ok=True)
    (root / "params.py").write_text(TEMPLATE.format(du_step=new_value))
    # copy any modules that must exist in sandbox import path
    # (optional): shutil.copytree("dawn_core", root, dirs_exist_ok=True)
    return PatchResult(run_id, str(root.parent), True, "")

# dawn_core/self_mod/sandbox_runner.py
import json, os, subprocess, sys, pathlib

def run_sandbox(sandbox_dir: str, ticks: int = 30) -> dict:
    env = os.environ.copy()
    project = str(pathlib.Path(".").resolve())
    env["PYTHONPATH"] = f"{sandbox_dir}:{project}:{env.get('PYTHONPATH','')}"
    code = """
import json
from dawn_core.state import set_state, get_state, label_for
from dawn_core.tick_orchestrator import demo_step
from dawn_core.params import params

set_state(unity=0.60, awareness=0.60, momentum=0.01, level='coherent', ticks=0)
start = get_state().unity
for _ in range(%(ticks)d):
    s = get_state()
    u = min(1.0, s.unity + demo_step())
    a = min(1.0, s.awareness + demo_step())
    set_state(unity=u, awareness=a, level=label_for(u,a), ticks=s.ticks+1)
end = get_state().unity
print(json.dumps({"ticks": %(ticks)d, "du_step": params.DU_STEP, "delta_unity": end - start, "end_unity": end, "end_level": get_state().level}))
""" % {"ticks": ticks}
    p = subprocess.run([sys.executable, "-c", code], env=env, capture_output=True, text=True)
    if p.returncode != 0:
        return {"ok": False, "error": p.stderr}
    return {"ok": True, "result": json.loads(p.stdout)}

E) Policy gate really compares baseline vs sandbox

# dawn_core/self_mod/policy_gate.py
from dataclasses import dataclass

@dataclass
class GateDecision:
    accept: bool
    reason: str

def decide(baseline: dict, sandbox: dict) -> GateDecision:
    if not baseline.get("ok"):  return GateDecision(False, f"baseline error: {baseline.get('error')}")
    if not sandbox.get("ok"):   return GateDecision(False, f"sandbox error: {sandbox.get('error')}")
    b = baseline["result"]; s = sandbox["result"]
    if s["end_unity"] < 0.85:   return GateDecision(False, "unity below safety floor")
    if s["delta_unity"] < b["delta_unity"] + 0.01: return GateDecision(False, "no meaningful improvement")
    if s["end_level"] not in ("meta_aware","transcendent"): return GateDecision(False, f"end level inadequate: {s['end_level']}")
    return GateDecision(True, "passes thresholds")
