# runners/dawn_run.py
import os, signal, time
from rich.live import Live
from rich.panel import Panel
from rich.table import Table

os.environ.setdefault("DAWN_DEMO_MODE", "1")

from dawn_core.state import get_state, set_state, label_for
from dawn_core.tick_orchestrator import demo_step
from scripts.health_check import main as health_check  # if you have it
# optional: from dawn_core.hub import CoordinationHub

STOP = False
def _sigint(*_): 
    global STOP; STOP = True

def render_status():
    s = get_state()
    t = Table.grid(expand=True)
    t.add_row(f"[bold]Ticks[/]: {s.ticks}",
              f"[bold]Level[/]: {s.level}",
              f"[bold]Unity[/]: {s.unity:.3f}",
              f"[bold]Awareness[/]: {s.awareness:.3f}")
    t.add_row(f"SCUP: {getattr(s,'scup',0):.3f}",
              f"SHI: {getattr(s,'shi',0):.3f}",
              f"Entropy: {getattr(s,'entropy',0):.3f}",
              f"DU_STEP: {getattr(__import__('dawn_core.params','fromlist',['params']).params,'DU_STEP',0.03):.3f}")
    return Panel(t, title="DAWN — Headless Console", border_style="cyan")

def main():
    # 0) health
    try: health_check()
    except Exception as e:
        print(f"⚠️ Health check warnings: {e}")

    # 1) reset state
    set_state(unity=0.00, awareness=0.00, momentum=0.01, level="fragmented", ticks=0)

    # 2) loop
    signal.signal(signal.SIGINT, _sigint)
    with Live(render_status(), refresh_per_second=8) as live:
        while not STOP:
            s = get_state()
            # update metrics
            u = min(1.0, s.unity + demo_step())
            a = min(1.0, s.awareness + demo_step())
            # simple EWMA momentum
            m = 0.9*s.momentum + 0.1*abs((u+a)-(s.unity+s.awareness))
            set_state(unity=u, awareness=a, momentum=m, level=label_for(u,a), ticks=s.ticks+1)

            # celebrate thresholds once (store flags on state if you support it)
            if u>=.60 and a>=.60 and s.level=="coherent": print("🟦 Coherent reached")
            if u>=.80 and a>=.80 and s.level=="meta_aware": print("🟪 Meta-aware achieved")
            if u>=.90 and a>=.90 and s.level=="transcendent": print("🌠 Transcendent achieved")

            # optional: trigger hub/self-mod every 50 ticks
            # if s.ticks % 50 == 0: maybe_hub_cycle()

            # refresh panel every 5 ticks
            if s.ticks % 5 == 0: live.update(render_status())
            time.sleep(0.02)  # throttle
    s = get_state()
    print(f"\n✅ Stopped. Final: level={s.level} unity={s.unity:.3f} awareness={s.awareness:.3f} ticks={s.ticks}")

if __name__ == "__main__":
    main()
