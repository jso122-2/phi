Types (Zod)

import { z } from "zod";

export const RunSchema = z.object({
  id: z.string(),
  type: z.enum(["docs","train","export"]),
  status: z.enum(["queued","running","success","failed","cancelled"]),
  startedAt: z.string(),
  finishedAt: z.string().nullable(),
  metrics: z.object({
    ticks: z.number().optional(),
    entropy: z.number().optional(),
    scup: z.number().optional(),
  }).partial(),
  logsUrl: z.string().url().nullable(),
});
export type Run = z.infer<typeof RunSchema>;

export const ArtifactSchema = z.object({
  id: z.string(),
  type: z.enum(["model","report","zine","image","log"]),
  name: z.string(),
  createdAt: z.string(),
  size: z.number().optional(),
  status: z.enum(["ready","processing","failed"]),
  downloadUrl: z.string().url().nullable(),
});
export type Artifact = z.infer<typeof ArtifactSchema>;

API client pattern

// src/lib/api.ts
import axios from "axios";
import { z } from "zod";
import { RunSchema, ArtifactSchema } from "./schemas";

const api = axios.create({ baseURL: import.meta.env.VITE_API_URL || "http://localhost:8787/v1" });

export async function listRuns() {
  const { data } = await api.get("/runs");
  return z.array(RunSchema).parse(data);
}

export async function getRun(id: string) {
  const { data } = await api.get(`/runs/${id}`);
  return RunSchema.parse(data);
}

export async function startRun(type: "docs"|"train"|"export") {
  const { data } = await api.post("/runs", { type });
  return RunSchema.parse(data);
}

export async function listArtifacts() {
  const { data } = await api.get("/artifacts");
  return z.array(ArtifactSchema).parse(data);
}
