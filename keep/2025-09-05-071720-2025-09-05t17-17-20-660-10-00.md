4) Coordination Hub (single writer; quorum policy)

Prompt:

Create dawn_core/hub.py. The hub requests proposals, scores them, and sends the winner through your existing self-mod sandbox (snapshot→sandbox→gate→promote). Only the hub mutates state via that path.

# dawn_core/hub.py
from typing import List, Dict, Any
from dataclasses import dataclass
from dawn_core.bubbles.base import Proposal, Bubble
from dawn_core.snapshots.read_only import capture_readonly_state
from dawn_core.self_mod.patch_builder import make_sandbox_du_step
from dawn_core.self_mod.sandbox_runner import run_sandbox, run_baseline
from dawn_core.self_mod.policy_gate import decide
from dawn_core.self_mod.promote import promote_du_step

@dataclass
class HubResult:
    action: str          # "NOOP" | "APPLY" | "REJECT"
    reason: str
    proposal: Proposal | None
    baseline: Dict[str, Any] | None
    sandbox: Dict[str, Any] | None

class CoordinationHub:
    def __init__(self, bubbles: List[Bubble], min_gain=0.01, floor=0.85):
        self.bubbles = bubbles
        self.min_gain = min_gain
        self.floor = floor

    def score(self, p: Proposal) -> float:
        return p.est_delta_unity + p.est_delta_shi + p.est_entropy_var_drop - 0.001*p.cost_ms

    def cycle(self) -> HubResult:
        s = capture_readonly_state()
        proposals: List[Proposal] = []
        for b in self.bubbles:
            try:
                p = b.think(s)
                if p: proposals.append(p)
            except Exception as e:
                print(f"[HUB] Bubble {b.name} error: {e}")
        if not proposals:
            return HubResult("NOOP","no proposals",None,None,None)

        winner = max(proposals, key=self.score)
        # Only DU_STEP constant supported in first pass
        if winner.kind != "constant" or winner.details.get("param") != "DU_STEP":
            return HubResult("REJECT","unsupported proposal type",winner,None,None)

        baseline = run_baseline(ticks=30)
        patch = make_sandbox_du_step(float(winner.details["new"]))
        sandbox = run_sandbox(patch.sandbox_dir, ticks=30)
        decision = decide(baseline, sandbox)
        if not decision.accept:
            return HubResult("REJECT", decision.reason, winner, baseline, sandbox)

        # Promote change live
        record = {
            "proposal": winner.__dict__,
            "baseline": baseline,
            "sandbox": sandbox,
            "reason": "policy accepted"
        }
        promote_du_step(float(winner.details["new"]), record)
        return HubResult("APPLY","applied safely", winner, baseline, sandbox)

def run_baseline(ticks: int = 30) -> dict:
    import json, subprocess, sys
    code = """
import json
from dawn_core.state import set_state, get_state, label_for
from dawn_core.tick_orchestrator import demo_step
from dawn_core.params import params
set_state(unity=0.60, awareness=0.60, momentum=0.01, level='coherent', ticks=0)
start = get_state().unity
for _ in range(%(ticks)d):
    s=get_state()
    u=min(1.0, s.unity + demo_step()); a=min(1.0, s.awareness + demo_step())
    set_state(unity=u, awareness=a, level=label_for(u,a), ticks=s.ticks+1)
end = get_state().unity
print(json.dumps({"ticks": %(ticks)d, "du_step": params.DU_STEP, "delta_unity": end-start, "end_unity": end, "end_level": get_state().level}))
""" % {"ticks": ticks}
    p = subprocess.run([sys.executable, "-c", code], capture_output=True, text=True)
    if p.returncode != 0:
        return {"ok": False, "error": p.stderr}
    return {"ok": True, "result": json.loads(p.stdout)}
