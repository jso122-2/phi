# validate_imports.py
"""
Walks the DAWN repo, tries to import every .py file, and reports:
✅ success | ❌ fail with error
"""

import os, importlib.util, traceback

def validate_imports(root=".", exclude=("venv", "__pycache__", "tests")):
    failures = []
    for dirpath, _, files in os.walk(root):
        if any(skip in dirpath for skip in exclude):
            continue
        for file in files:
            if file.endswith(".py"):
                module_path = os.path.join(dirpath, file)
                rel_path = os.path.relpath(module_path, root).replace("/", ".")[:-3]
                try:
                    spec = importlib.util.spec_from_file_location(rel_path, module_path)
                    mod = importlib.util.module_from_spec(spec)
                    spec.loader.exec_module(mod)
                    print(f"✅ {rel_path}")
                except Exception:
                    print(f"❌ {rel_path}")
                    traceback.print_exc()
                    failures.append(rel_path)
    if failures:
        raise SystemExit(f"\n{len(failures)} import failures: {failures}")

if __name__ == "__main__":
    validate_imports("dawn_core")

# fix_imports.py
"""
Scans imports in DAWN .py files.
If a relative import fails, rewrite it as package-relative (dawn_core.*).
Dry-run first, then apply fixes.
"""

import re, pathlib

root = pathlib.Path("dawn_core")
pattern = re.compile(r"^(from\s+)(\w+)(\s+import)", re.M)

for py in root.rglob("*.py"):
    txt = py.read_text()
    fixed = pattern.sub(r"\1dawn_core.\2\3", txt)
    if fixed != txt:
        print(f"Patched: {py}")
        py.write_text(fixed)

3. Circular Import Detector

# detect_circulars.py
"""
Builds a dependency graph of imports inside dawn_core.
Detects circular dependencies and prints the loop chain.
"""

import os, ast, networkx as nx

def get_imports(path):
    with open(path, "r") as f:
        tree = ast.parse(f.read())
    return [
        node.module for node in ast.walk(tree)
        if isinstance(node, ast.ImportFrom) and node.module
    ]

graph = nx.DiGraph()
root = "dawn_core"

for dirpath, _, files in os.walk(root):
    for f in files:
        if f.endswith(".py"):
            fp = os.path.join(dirpath, f)
            mod = fp.replace("/", ".")[:-3]
            for dep in get_imports(fp):
                if dep.startswith("dawn_core"):
                    graph.add_edge(mod, dep)

cycles = list(nx.simple_cycles(graph))
if cycles:
    print("❌ Circular imports detected:")
    for cycle in cycles:
        print(" -> ".join(cycle))
else:
    print("✅ No circular imports found")
