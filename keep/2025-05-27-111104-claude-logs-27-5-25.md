@dataclass
class DriftVector:
    """Represents semantic drift with inheritance tracking"""
    angle: float
    magnitude: float
    generation: int
    parent_id: Optional[str] = None
    lineage_chain: List[str] = field(default_factory=list)
    timestamp: float = field(default_factory=time.time)
    
    def compute_drift_from_parent(self, parent_vector: 'DriftVector') -> 'DriftVector':
        """Compute drift relative to parent, preserving inheritance"""
        new_angle = self.angle + (parent_vector.angle * 0.3)  # Partial inheritance
        new_magnitude = math.sqrt(self.magnitude**2 + parent_vector.magnitude**2)
        
        return DriftVector(
            angle=new_angle,
            magnitude=new_magnitude,
            generation=parent_vector.generation + 1,
            parent_id=parent_vector.parent_id,
            lineage_chain=parent_vector.lineage_chain + [str(parent_vector.parent_id)]
        )

@dataclass
class Sigil:
    """Basic semantic unit with lifecycle"""
    id: str
    entropy: float
    age: int = 0
    decay_rate: float = 0.02
    generation: int = 0
    ancestry: List[str] = field(default_factory=list)
    semantic_vector: List[float] = field(default_factory=lambda: [random.random() for _ in range(8)])
    
    def decay(self):
        self.entropy *= (1 - self.decay_rate)
        self.age += 1
    
    def is_expired(self) -> bool:
        return self.entropy < 0.1 or self.age > 100

class PulseHeat:

        self.tick_count = 0
        self.pressure_history = deque(maxlen=50)
        self.heat_history = deque(maxlen=100)

    def update(self, pressure: float):
        self.pressure_history.append(pressure)
        self.heat_history.append(self.heat)
        self.tick_count += 1
        
        # Adaptive heat regulation
        if pressure > 0.7:
            self.heat *= 1.1  # Increase heat under pressure
        elif pressure < 0.3:
            self.heat *= 0.95  # Cool down when calm
            
        self.heat = max(0.1, min(5.0, self.heat))  # Bounds
    
    def get_heat(self) -> float:
        return self.heat
    
    def boost(self, amount: float):
        self.heat += amount
    
    def apply_penalty(self, seed_id: str, penalty: float):
        self.heat *= (1 - penalty * 0.1)  # Reduce heat as penalty
