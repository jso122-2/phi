class ChangeDetector:
    """
    The simplest rod: just checks if pixels changed.
    
    Like a security camera motion detector:
    - If many pixels change = motion detected
    - If few pixels change = nothing happening
    """
    
    def __init__(self, sensitivity: float = 0.1):
        self.sensitivity = sensitivity  # How much change to care about
        self.previous_frame = None
    
    def detect(self, current_frame: np.ndarray) -> ChangeEvent:
        """
        Compare current frame to previous frame.
        Return WHERE things changed and HOW MUCH.
        """
        
        if self.previous_frame is None:
            # First frame, nothing to compare
            self.previous_frame = current_frame
            return ChangeEvent(changed=False, regions=[])
        
        # 1. Calculate difference
        diff = cv2.absdiff(self.previous_frame, current_frame)
        
        # 2. Convert to grayscale (simpler)
        gray_diff = cv2.cvtColor(diff, cv2.COLOR_BGR2GRAY)
        
        # 3. Threshold: only care about big changes
        _, thresh = cv2.threshold(
            gray_diff, 
            int(255 * self.sensitivity),  # Min change to detect
            255, 
            cv2.THRESH_BINARY
        )
        
        # 4. Find regions that changed (contours)
        contours, _ = cv2.findContours(
            thresh, 
            cv2.RETR_EXTERNAL, 
            cv2.CHAIN_APPROX_SIMPLE
        )
        
        # 5. Build list of change regions
        changed_regions = [
            ChangeRegion(
                bbox=cv2.boundingRect(contour),
                area=cv2.contourArea(contour),
                intensity=np.mean(gray_diff[mask])
            )
            for contour in contours
            if cv2.contourArea(contour) > 100  # Ignore tiny changes
        ]
        
        # 6. Update for next time
        self.previous_frame = current_frame
        
        return ChangeEvent(
            changed=len(changed_regions) > 0,
            regions=changed_regions
        )

class RodValidator:
    """
    The rod system manager.
    
    Runs all rods in sequence, collects results.
    """
    
    def __init__(self):
        self.motion_rod = MotionRod()
        self.template_rod = TemplateRod(load_templates())
        self.spatial_rod = SpatialRod(load_spatial_rules())
        self.temporal_rod = TemporalRod()
    
    def validate(self, frame: np.ndarray, 
                 detection: Detection) -> ValidationResult:
        """
        Run all rods, collect results.
        
        If ANY rod fails → validation fails.
        """
        
        results = []
        
        # 1. Check motion (did something change?)
        motion_result = self.motion_rod.validate(frame)
        results.append(('motion', motion_result))
        
        # 2. Check template (does it look right?)
        template_result = self.template_rod.validate(
            frame, detection.object_type
        )
        results.append(('template', template_result))
        
        # 3. Check spatial (is it in right place?)
        spatial_result = self.spatial_rod.validate(detection)
        results.append(('spatial', spatial_result))
        
        # 4. Check temporal (is transition legal?)
        temporal_result = self.temporal_rod.validate(detection.state)
        results.append(('temporal', temporal_result))
        
        # 5. Aggregate results
        all_passed = all(r.passed for _, r in results)
        avg_confidence = np.mean([r.confidence for _, r in results])
        
        return ValidationResult(
            validated=all_passed,
            confidence=avg_confidence,
            rod_results=dict(results)
        )
